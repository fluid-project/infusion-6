// Combines the image generator with the closest-note finder: the text of the sticky note nearest to the widget
// becomes the image prompt, so moving the widget next to a note and pressing the button generates an image of it
fluid.def("fluid.demos.imageFromClosestNote", {
    $layers: ["fluid.demos.imageGenerator", "fluid.topLevelWidget"],
    // Finds the closest note silently: it renders nothing, borrowing this widget's own view
    finder: {
        $component: {
            $layers: "fluid.demos.closestNoteFinder",
            view: "{imageFromClosestNote}.view"
        }
    },
    // The closest note's text is the prompt, re-read whenever the widget moves
    state: {
        prompt: "{finder}.noteText"
    },
    buttonLabel: {
        $compute: {
            func: "fluid.demos.imageFromClosestNote.buttonLabel",
            args: ["{self}.isGenerating"]
        }
    }
});

fluid.demos.imageFromClosestNote.buttonLabel = isGenerating =>
    isGenerating ? "Finding..." : "Generate image of closest note";
