<script>
// Combines fluflu.demos.closestNoteFinder with fluid.demos.imageGenerator: the text of the sticky note
// closest to the widget becomes the prompt for the image search, so the widget can be dragged near a
// note and its "Find Image" button pressed.
fluid.def("fluid.demos.noteDrivenImageGenerator", {
    $layers: "fluid.topLevelWidget",
    preferredSize: {
        w: 360,
        h: 640
    },
    // Reports on the note nearest to this widget's own tldraw shape - the finder's topLevelWidget-issued
    // hostBounds cannot see our shape, so it is bound to ours
    finder: {
        $component: {
            $layers: "fluid.demos.closestNoteFinder",
            hostBounds: "{noteDrivenImageGenerator}.hostBounds"
        }
    },
    generator: {
        $component: {
            $layers: "fluid.demos.imageGenerator"
        }
    },
    // The closest note's text becomes the image search prompt, kept in step as the widget is moved
    syncImagePrompt: {
        $effect: {
            func: (state, prompt) => {
                state.prompt = prompt;
            },
            args: ["{generator}.state", "{finder}.closest.text"]
        }
    }
});
</script>

<template>
    <div class="noteimage">
        <div @id="finder" class="noteimage-finder"></div>
        <div @id="generator" class="noteimage-generator"></div>
    </div>
</template>
