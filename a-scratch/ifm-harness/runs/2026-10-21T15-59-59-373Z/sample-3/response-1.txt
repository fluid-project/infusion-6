<script>
// Composition of the closest-note finder and the Pixabay image generator. The finder's closest-note machinery is
// inherited at `{self}.closest`, the generator's search machinery at `{self}.state`; a button joins them, seeding
// the prompt with the closest sticky note's text and starting the search when pressed.

fluid.def("fluid.demos.noteImageGenerator", {
    $layers: ["fluid.demos.closestNoteFinder", "fluid.demos.imageGenerator", "fluid.topLevelWidget"],
    preferredSize: {
        w: 420,
        h: 680
    },
    // Invoked by the button: take the closest note's text as the prompt and start the search
    generateFromNote: {
        $method: {
            func: "fluid.demos.noteImageGenerator.generateFromNote",
            args: ["{self}.closest", "{self}.state"]
        }
    }
});

/**
 * Seed the image generator's prompt with the closest sticky note's text and start searching for it
 * @param {ClosestNoteRec} closest - Description of the closest sticky note, or an unavailable value
 * @param {Object} state - The component's reactive generator state
 * @return {Promise|undefined} The search promise once it has started
 */
fluid.demos.noteImageGenerator.generateFromNote = function (closest, state) {
    if (closest && closest.distance !== null) {
        state.prompt = closest.text;
        return fluid.demos.imageGenerator.generate(state);
    }
    state.error = "Move the widget next to a sticky note to use its words as a prompt";
    state.images = [];
    state.index = 0;
    state.isGenerating = false;
};
</script>

<template>
    <div class="note-imagegen">
        <div class="note-imagegen-finder">
            <h3 class="note-imagegen-title">Closest Sticky Note</h3>
            <div class="note-imagegen-text">@{noteText}</div>
            <div @id="distanceLine"></div>
        </div>
        <div class="note-imagegen-divider"></div>
        <div class="note-imagegen-generator">
            <p class="note-imagegen-prompt">Prompt: “@{state.prompt}”</p>
            <div class="note-imagegen-buttons">
                <button class="note-imagegen-button" disabled="@{isGenerating}"
                        @onclick="{self}.generateFromNote()">Image from this note</button>
                <button class="note-imagegen-button note-imagegen-another" disabled="@{cannotShowAnother}"
                        @onclick="{self}.showAnother()">Another</button>
            </div>
            <div @id="errorMessage"></div>
            <div @id="loadingMessage"></div>
            <div @id="result"></div>
        </div>
    </div>
</template>
