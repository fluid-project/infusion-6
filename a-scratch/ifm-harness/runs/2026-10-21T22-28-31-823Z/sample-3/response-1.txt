<script>
// Composes fluid.demos.closestNoteFinder with fluid.demos.imageGenerator: the widget watches
// the closest sticky note on the canvas, and pressing the button runs an image search for that
// note's text.

fluid.def("fluid.demos.noteImageGen", {
    $layers: "fluid.topLevelWidget",
    preferredSize: {
        w: 400,
        h: 700
    },
    noteFinder: {
        $component: {
            $layers: "fluid.demos.closestNoteFinder"
        }
    },
    imageGen: {
        $component: {
            $layers: "fluid.demos.imageGenerator"
        }
    },
    generateFromClosest: {
        $method: {
            func: "fluid.demos.noteImageGen.generateFromClosest",
            args: ["{self}.noteFinder", "{self}.imageGen"]
        }
    }
});

/**
 * Set the image generator's prompt to the closest note's text, and run the search for it.
 * The search's promise is returned so the whole operation is a single undoable step.
 * @param {Component} noteFinder - The closest-note finder subcomponent
 * @param {Component} imageGen - The image generator subcomponent
 */
fluid.demos.noteImageGen.generateFromClosest = function (noteFinder, imageGen) {
    const record = noteFinder.closest;
    if (record.distance === null) {
        imageGen.state.error = "Move this widget next to a sticky note, then press Generate again";
        return;
    }
    imageGen.state.prompt = record.text;
    return imageGen.generate();
};
</script>

<template>
    <div class="noteimg">
        <!-- The closest-note finder renders its own summary and distance line -->
        <div @id="noteFinder"></div>
        <!-- One press takes the closest note's text straight to the image generator -->
        <button class="noteimg-generate" @onclick="{self}.generateFromClosest()">Generate an image for this note</button>
        <!-- The full image generator (prompt, results and credits) -->
        <div @id="imageGen"></div>
    </div>
</template>
