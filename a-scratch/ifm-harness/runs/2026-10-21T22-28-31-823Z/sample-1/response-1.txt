<script>
// Combines fluid.demos.closestNoteFinder and fluid.demos.imageGenerator: drag the widget next to a sticky note
// and press "Generate Image" to have Pixabay generate an image from the note's text.
fluid.def("fluid.demos.noteImageGen", {
    $layers: "fluid.topLevelWidget",
    preferredSize: {
        w: 400,
        h: 720
    },
    // The finder computes the note closest to OUR bounds - it has no tldraw shape of its own, so its own
    // hostBounds would come to nothing and the note text would never be found
    finder: {
        $component: {
            $layers: "fluid.demos.closestNoteFinder",
            hostBounds: "{noteImageGen}.hostBounds"
        }
    },
    imageGen: {
        $component: {
            $layers: "fluid.demos.imageGenerator"
        }
    },
    generateFromNote: {
        $method: {
            func: "fluid.demos.noteImageGen.generateFromNote",
            args: ["{self}"]
        }
    }
});

/**
 * Put the closest note's text into the image generator's prompt and start a search
 * @param {Component} self - The umbrella component, proxy to which the method is invoked
 */
fluid.demos.noteImageGen.generateFromNote = function (self) {
    const imageGen = self.imageGen;
    let closest;
    try {
        closest = self.finder.closest;
    } catch (unavailable) {
        closest = null;
    }
    if (!closest || closest.distance === null) {
        imageGen.state.error = "Move me next to a sticky note to use its text";
    } else if (!closest.text || closest.text === "Empty note") {
        imageGen.state.error = "That sticky note is empty - write a prompt in it first";
    } else {
        imageGen.state.prompt = closest.text;
        imageGen.generate();
    }
};
</script>

<template>
    <div class="noteimagegen">
        <h3 class="noteimagegen-title">Sticky Note Image Generator</h3>
        <!-- The finder renders the closest note's text and, while it finds one, its distance -->
        <div @id="finder"></div>
        <button class="noteimagegen-button" @onclick="{self}.generateFromNote()">Generate Image</button>
        <!-- The full image generator: prompt input, its own Generate/Another buttons, result and messages -->
        <div @id="imageGen"></div>
    </div>
</template>
