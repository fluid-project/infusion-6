// A wrapper combining the closest-note finder and the image generator: it shows the sticky note nearest
// to the widget on the canvas, and generates an image from that note's text when asked. Move the widget
// close to a note to load the note's text into the prompt, then press "Find Image", or edit the prompt
// first.

fluid.def("fluid.demos.stickyNoteImageGen", {
    $layers: ["fluid.demos.closestNoteFinder", "fluid.demos.imageGenerator", "fluid.topLevelWidget"],
    preferredSize: {
        w: 420,
        h: 560
    },
    // As the closest note changes, put its text into the generator's prompt so it can be searched as-is.
    // The prompt may still be edited: updates sourced from the DOM are ignored until the closest note
    // changes again. The guard keeps the write converge, since writing the prompt re-notifies this effect.
    syncPrompt: {
        $effect: {
            func: (state, text, distance) => {
                if (distance !== null && state.prompt !== text) {
                    state.prompt = text;
                }
            },
            args: ["{self}.state", "{self}.noteText", "{self}.distance"],
            excludeSource: "dom"
        }
    },
    view: {
        $component: {
            $layers: "fluid.templateViewComponent",
            template: `<div style="width:100%;height:100%;box-sizing:border-box;display:flex;flex-direction:column;padding:16px;background:#fffde7;font-family:sans-serif;color:#333">
    <div style="display:flex;flex-direction:column;flex:none;margin-bottom:12px">
        <h3 style="margin:0 0 8px 0;font-size:14px;color:#666">Closest Sticky Note</h3>
        <div style="flex:1;display:flex;align-items:center;justify-content:center;font-size:18px;font-weight:bold;text-align:center">@{noteText}</div>
        <div @id="distanceLine"></div>
    </div>
    <div style="display:flex;flex:none;gap:8px;margin-bottom:12px">
        <input type="text" style="flex:1;padding:10px 12px;font:inherit;border:1px solid #ccc;box-sizing:border-box" placeholder="Describe the image you want..." value="@{state.prompt}" disabled="@{isGenerating}" @onkeydown="{self}.keyDown({0})">
    </div>
    <div style="display:flex;flex:none;gap:8px;margin-bottom:12px">
        <button style="flex:1;padding:10px 16px;font:inherit;font-weight:500;color:#fff;border:none;background:#4a90d9;cursor:pointer" disabled="@{cannotGenerate}" @onclick="{self}.generate()">@{buttonLabel}</button>
        <button style="padding:10px 16px;font:inherit;font-weight:500;color:#fff;border:none;background:#607d8b;cursor:pointer" disabled="@{cannotShowAnother}" @onclick="{self}.showAnother()">Another</button>
    </div>
    <div @id="errorMessage"></div>
    <div @id="loadingMessage"></div>
    <div @id="result" style="flex:1 1 auto;min-height:0;display:flex;flex-direction:column;margin-top:12px"></div>
</div>`
        }
    }
});
