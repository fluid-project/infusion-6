fluid.demos.ai.noteToImage.hasNoNote = closest => !(closest && closest.distance !== null);

fluid.demos.ai.noteToImage.generateFromNote = function (imageGen, closest) {

        imageGen.state.prompt = closest.text;
        return imageGen.generate();
