// Composes the closest-note finder with the image generator: the finder runs hidden behind
// the image generator's view, and its closest note's text becomes the generator's prompt.
// The "Find Image" button is only enabled while the widget sits near an actual note.

fluid.def("fluid.demos.ai.imageFromClosestNote", {
    $layers: ["fluid.demos.imageGenerator", "fluid.topLevelWidget"],
    noteFinder: {
        $component: {
            $layers: "fluid.demos.closestNoteFinder",
            // Share the image generator's own view, so the finder computes without rendering
            view: "{imageFromClosestNote}.view"
        }
    },
    // Only a note which is actually nearby supplies a prompt - the finder's strings for
    // "not on a canvas" or "no notes" are display text, not search prompts
    nearbyNoteText: "$compute:fluid.demos.ai.imageFromClosestNote.notePrompt({noteFinder}.noteText, {noteFinder}.distance)",
    state: {
        prompt: "{self}.nearbyNoteText"
    }
});

fluid.demos.ai.imageFromClosestNote.notePrompt = (noteText, distance) => distance !== null ? noteText : "";
