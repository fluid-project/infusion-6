<script>
// Umbrella layer combining the two selected widgets: the closest-note finder shows the sticky note
// nearest to the widget (recomputed as it is moved on the canvas), and a single button copies that
// note's text into the image generator's prompt and starts the Pixabay search. Neither existing layer
// is modified - they are composed as subcomponents, with `closest` overridden so that our host bounds
// drive it (hostBounds is supplied to the hosted top-level widget, not to a subcomponent).

fluid.def("fluid.demos.closestImageGenerator", {
    $layers: "fluid.topLevelWidget",
    preferredSize: {
        w: 380,
        h: 720
    },
    noteFinder: {
        $component: {
            $layers: "fluid.demos.closestNoteFinder",
            closest: "$compute:fluid.demos.closestNoteFinder.findClosest({tldrawHost}.notes, {closestImageGenerator}.hostBounds)"
        }
    },
    imageGen: {
        $component: {
            $layers: "fluid.demos.imageGenerator"
        }
    },
    generateFromNote: {
        $method: {
            func: "fluid.demos.closestImageGenerator.generateFromNote",
            args: ["{noteFinder}:finder", "{imageGen}:generator"]
        }
    }
});

/**
 * Use the closest note as the image prompt: copy its text into the generator's prompt and start
 * the search, guiding the user if there is no usable note nearby.
 * @param {Component} finder - The closest-note finder subcomponent
 * @param {Component} generator - The image generator subcomponent
 */
fluid.demos.closestImageGenerator.generateFromNote = function (finder, generator) {
    const closest = finder.closest;
    if (!closest || closest.distance === null) {
        generator.state.error = "No sticky note nearby - move the widget next to one, then press the button again";
    } else if (closest.text === "Empty note") {
        generator.state.error = "The closest sticky note is empty - put some words in it first";
    } else {
        generator.state.prompt = closest.text;
        generator.generate();
    }
};
</script>

<template>
    <div class="closest-imagegen">
        <div class="closest-imagegen-finder" @id="noteFinder"></div>
        <button class="closest-imagegen-button" disabled="@{imageGen.isGenerating}"
                @onclick="{self}.generateFromNote()">Generate image of the closest note</button>
        <div class="closest-imagegen-gen" @id="imageGen"></div>
    </div>
</template>
