<script>
fluid.def("fluid.demos.noteImageGenerator", {
    $layers: "fluid.topLevelWidget",
    preferredSize: {w: 660, h: 480},
    // The note finder finds the sticky note nearest THIS widget, so it must read the umbrella's bounds
    // rather than host bounds of its own, which it no longer has as a subcomponent
    noteFinder: {
        $component: {
            $layers: "fluid.demos.closestNoteFinder",
            hostBounds: "{noteImageGenerator}.hostBounds"
        }
    },
    // The image generator searches for images of the nearest note's text, so moving this widget close to
    // a sticky note fills the prompt and pressing its Find Image button generates an image of the note
    imageGen: {
        $component: {
            $layers: "fluid.demos.imageGenerator",
            state: {
                prompt: "{noteImageGenerator}.promptText"
            }
        }
    },
    // Blank prompt while no note is in reach, so the button can't fire pointlessly
    promptText: "$compute:fluid.demos.noteImageGenerator.promptText({noteFinder}.noteText, {noteFinder}.distance)"
});

fluid.demos.noteImageGenerator.promptText = (noteText, distance) => distance === null ? "" : noteText;
</script>
