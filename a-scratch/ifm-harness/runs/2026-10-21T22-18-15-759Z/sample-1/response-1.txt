<script>
// Combines the closest-note finder with the image generator. The finder tracks the sticky note nearest
// to this widget's own shape, and the generator's "Find Image" button takes its prompt from that note's
// text - so the flow is: move the widget close to a note, press the button, get an image of the note's subject.

fluid.def("fluid.demos.noteDrivenImageGen", {
    $layers: "fluid.topLevelWidget",
    preferredSize: {
        w: 380,
        h: 680
    },
    // Track the note closest to this widget, using the bounds of the combined widget's own shape
    finder: {
        $component: {
            $layers: "fluid.demos.closestNoteFinder",
            // topLevelWidget feeds bounds only to the root component, so hand them down to the finder
            hostBounds: "{noteDrivenImageGen}.hostBounds"
        }
    },
    // Generate an image of whatever the finder points at, rather than a prompt typed by hand
    imageGen: {
        $component: {
            $layers: "fluid.demos.imageGenerator",
            // The button is live whenever the finder has a note to work from
            cannotGenerate: "$compute:fluid.demos.noteDrivenImageGen.cannotGenerate({self}.isGenerating, {finder}.distance)",
            // Seed the prompt with the closest note's text, then run the usual Pixabay search
            generate: {
                $method: {
                    func: "fluid.demos.noteDrivenImageGen.generate",
                    args: ["{self}.state", "{finder}.closest"]
                }
            }
        }
    }
});

/**
 * Enable the button whenever we are not already searching and a sticky note is in range
 * @param {Boolean} isGenerating - Whether a search is already under way
 * @param {Number|null} distance - Distance to the closest note, or `null` if there is none
 * @return {Boolean} `true` to disable the button
 */
fluid.demos.noteDrivenImageGen.cannotGenerate = (isGenerating, distance) => isGenerating || distance === null;

/**
 * Search Pixabay for an image matching the text of the closest sticky note
 * @param {Object} state - The image generator's reactive state
 * @param {Object} closest - The finder's closest note record ({text, distance})
 */
fluid.demos.noteDrivenImageGen.generate = async function (state, closest) {
    if (!closest || closest.distance === null) {
        state.error = "Move the widget close to a sticky note, then press Find Image";
    } else {
        state.prompt = closest.text;
        await fluid.demos.imageGenerator.generate(state);
    }
};
</script>

<template>
    <div class="note-driven">
        <div @id="finder"></div>
        <div class="note-driven-gap"></div>
        <div @id="imageGen"></div>
    </div>
</template>
