// Combines the image generator (dominant) with the closest note finder (hidden supplier):
// the prompt bound to the generator's input - and acted on by its "Find Image" button - is
// replaced by the text of the sticky note closest to the widget, wherever it is placed.
fluid.def("fluid.demos.ai.noteImageGenerator", {
    $layers: ["fluid.demos.imageGenerator", "fluid.topLevelWidget"],
    // Hidden supplier: shares this shape's bounds and view, so it renders nothing of its own
    finder: {
        $component: {
            $layers: "fluid.demos.closestNoteFinder",
            view: "{noteImageGenerator}.view",
            hostBounds: "{noteImageGenerator}.hostBounds"
        }
    },
    // The member takes the note's text whenever the widget moves near a different note
    state: {
        prompt: "{finder}.noteText"
    }
});
