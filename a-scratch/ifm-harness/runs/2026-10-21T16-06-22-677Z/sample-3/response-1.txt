// Combines the image generator (dominant) with the closest-note finder (supplier).
// The finder is hidden: it shares this widget's view, so its computation runs while
// nothing of its own renders. Its closest note's text becomes the prompt; with no
// note found the prompt is empty and the button stays disabled.

fluid.def("fluid.demos.imageGenFromClosestNote", {
    $layers: ["fluid.demos.imageGenerator", "fluid.topLevelWidget"],
    finder: {
        $component: {
            $layers: "fluid.demos.closestNoteFinder",
            view: "{imageGenFromClosestNote}.view",
            distanceLine: false
        }
    },
    promptText: "$compute:fluid.demos.imageGenFromClosestNote.promptOfClosest({finder}.closest)",
    state: {
        prompt: "{self}.promptText"
    }
});

fluid.demos.imageGenFromClosestNote.promptOfClosest = function (closest) {
    return !closest || closest.distance === null ? "" : closest.text;
};
