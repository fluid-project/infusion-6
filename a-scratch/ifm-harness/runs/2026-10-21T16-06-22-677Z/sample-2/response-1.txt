// Composes the image generator (dominant) with the closest note finder (supplier).
// The prompt is always the text of the sticky note nearest to wherever the widget is placed.
fluid.def("fluid.demos.ai.imageForClosestNote", {
    $layers: ["fluid.demos.imageGenerator", "fluid.topLevelWidget"],
    // The finder is hidden: its view is shared with this widget, so nothing of its own renders,
    // while its computation continues and tracks this widget's shape bounds.
    noteFinder: {
        $component: {
            $layers: "fluid.demos.closestNoteFinder",
            view: "{imageForClosestNote}.view"
        }
    },
    // The prompt follows the closest note's text, updating live as the widget moves.
    state: {
        prompt: "{noteFinder}.noteText"
    }
});
