fluid.def("fluid.demos.ai.noteImageGenerator", {
    $layers: ["fluid.demos.imageGenerator", "fluid.topLevelWidget"],
    // The closest note finder is the supplier: a hidden subcomponent sharing this widget's view, its output
    // wired into the image generator's prompt, so moving the widget near a sticky note fills the prompt
    finder: {
        $component: {
            $layers: "fluid.demos.closestNoteFinder",
            view: "{noteImageGenerator}.view",
            // Read the host bounds from the top-level widget, not from this subcomponent
            closest: "$compute:fluid.demos.closestNoteFinder.findClosest({tldrawHost}.notes, {noteImageGenerator}.hostBounds)"
        }
    },
    state: {
        prompt: "{finder}.closest.text"
    }
});
