// Combines fluid.demos.closestNoteFinder and fluid.demos.imageGenerator: the closest sticky note's text
// supplies the prompt for the image search. The finder is a hidden subcomponent whose view is shared with
// this widget, so only the image generator's template renders, while the finder's computation (its bounds
// and the notes held in tldrawHost) continues.
// Usage: drag the widget next to a sticky note - its text fills the prompt - and press "Find Image".

fluid.def("fluid.demos.ai.noteImageGenerator", {
    $layers: ["fluid.demos.imageGenerator", "fluid.topLevelWidget"],
    closestNoteFinder: {
        $component: {
            $layers: "fluid.demos.closestNoteFinder",
            view: "{noteImageGenerator}.view"
        }
    },
    // The closest note's text is always the prompt, so it follows the widget as it is moved around the canvas
    state: {
        prompt: "{closestNoteFinder}.noteText"
    }
});
