<script>
// Combines the closest note finder with the image generator: move this widget next to a sticky note and the note's
// text becomes the search prompt; press the button to generate an image of it
fluid.def("fluid.demos.noteImageGenerator", {
    $layers: "fluid.topLevelWidget",
    preferredSize: {
        w: 380,
        h: 680
    },
    finder: {
        $component: {
            $layers: ["fluid.demos.closestNoteFinder", "fluid.component"]
        }
    },
    imageGen: {
        $component: {
            $layers: ["fluid.demos.imageGenerator", "fluid.component"],
            // The prompt follows the closest note's text; typing in the box can still overwrite it
            state: {
                prompt: "{finder}.closest.text"
            }
        }
    }
});
</script>

<template>
    <div class="note-imagegen">
        <div class="note-imagegen-finder" @id="finder"></div>
        <div class="note-imagegen-imagegen" @id="imageGen"></div>
    </div>
</template>
