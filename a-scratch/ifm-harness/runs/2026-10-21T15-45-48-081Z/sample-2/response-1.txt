<script>
// Combines the closest-note finder with the Pixabay image generator: the generator seeks an image of the
// sticky note nearest the widget, so the user can move the widget next to a note and press "Find Image"

fluid.def("fluid.demos.noteImageGen", {
    $layers: "fluid.topLevelWidget",
    preferredSize: {
        w: 360,
        h: 700
    },
    // Which sticky note is nearest the widget, with the distance beneath
    closestNoteFinder: {
        $component: {
            $layers: "fluid.demos.closestNoteFinder"
        }
    },
    // The same generator, but its button is armed by the nearest note rather than a typed prompt
    imageGenerator: {
        $component: {
            $layers: "fluid.demos.imageGenerator",
            // Enabled while a search isn't running and the widget is beside a note (a distance of 0 still is)
            cannotGenerate: {
                $compute: {
                    func: (isGenerating, distance) => isGenerating || distance === null,
                    args: ["{self}.isGenerating", "{closestNoteFinder}.distance"]
                }
            },
            // Pressing the button seeks an image for the nearest note's text, and pre-fills the prompt
            generate: {
                $method: {
                    func: "fluid.demos.noteImageGen.generateFromNote",
                    args: ["{self}.state", "{closestNoteFinder}.noteText", "{closestNoteFinder}.distance"]
                }
            }
        }
    }
});

/**
 * Seek an image of the sticky note nearest the widget, using the note's text as the prompt
 * @param {Object} state - The image generator subcomponent's reactive state
 * @param {String} noteText - The text of the closest note
 * @param {Number|null} distance - The distance to the closest note, or `null` if there is none
 * @return {Promise|undefined} The search's promise, so that everything it causes is a single undoable step
 */
fluid.demos.noteImageGen.generateFromNote = function (state, noteText, distance) {
    if (distance === null) {
        state.error = "Move the widget next to a sticky note first";
        return undefined;
    }
    state.prompt = noteText;
    return fluid.demos.imageGenerator.generate(state);
};
</script>

<template>
    <div class="noteimagegen">
        <div class="noteimagegen-finder" @id="closestNoteFinder"></div>
        <div class="noteimagegen-generator" @id="imageGenerator"></div>
    </div>
</template>

    min-height: 0;
}
