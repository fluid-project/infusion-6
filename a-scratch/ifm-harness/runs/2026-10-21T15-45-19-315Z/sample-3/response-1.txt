// Combines the image generator with the closest note finder. The image generator is dominant - this widget is it,
// with all of its template, state and methods - while a hidden note finder keeps computing which sticky note is
// nearest and feeds its text to the generator's prompt, so moving the widget near a note pre-fills the search
fluid.def("fluid.demos.ai.noteFinderImageGenerator", {
    $layers: ["fluid.demos.imageGenerator", "fluid.topLevelWidget"],
    // The finder shares this widget's view, so neither it nor its subcomponents renders anything, whilst its
    // computation continues - its reading of the widget's position uses the same shape bounds as the widget itself
    finder: {
        $component: {
            $layers: "fluid.demos.closestNoteFinder",
            view: "{noteFinderImageGenerator}.view"
        }
    },
    // The prompt bound to the input, and so the search run by the button, is the nearest note's text, updated
    // as the widget is moved
    state: {
        prompt: "{finder}.closest.text"
    }
});
