fluid.def("fluid.demos.closestNoteImageGen", {
    $layers: ["fluid.demos.imageGenerator", "fluid.topLevelWidget"],
    // The finder supplies the prompt - the text of the nearest sticky note. Its view is redirected to
    // this widget's own, so it renders nothing while it keeps tracking the canvas position we share
    closestNoteFinder: {
        $component: {
            $layers: "fluid.demos.closestNoteFinder",
            view: "{closestNoteImageGen}.view"
        }
    },
    // The image generator's prompt follows the nearest note, staying blank while none is found
    closestNotePrompt: {
        $compute: {
            func: (noteText, distance) => distance === null ? "" : noteText,
            args: ["{closestNoteFinder}.noteText", "{closestNoteFinder}.distance"]
        }
    },
    state: {
        prompt: "{self}.closestNotePrompt"
    }
});
