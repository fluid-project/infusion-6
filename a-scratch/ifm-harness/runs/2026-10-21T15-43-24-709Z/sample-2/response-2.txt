<script>
// Combines the closest sticky note finder with the image generator: the widget shows the sticky note nearest to
// it, and pressing the button generates an image of that note, reusing the finder's note search and the generator's
// Pixabay search functions

fluid.def("fluid.demos.noteImageFinder", {
    $layers: "fluid.topLevelWidget",
    preferredSize: {
        w: 360,
        h: 480
    },
    // The closest note, exactly as the finder computes it
    closest: "$compute:fluid.demos.closestNoteFinder.findClosest({tldrawHost}.notes, {self}.hostBounds)",
    noteText: "{self}.closest.text",
    distance: "{self}.closest.distance",
    // As for the image generator, minus the prompt, which comes from the note
    state: {
        $reactiveRoot: {
            isGenerating: false,
            error: "",
            images: [],
            index: 0,
            refreshedId: null
        }
    },
    isGenerating: "{self}.state.isGenerating",
    error: "{self}.state.error",
    image: "$compute:fluid.demos.imageGenerator.currentImage({self}.state.images, {self}.state.index)",
    cannotGenerate: "$compute:fluid.demos.noteImageFinder.cannotGenerate({self}.distance, {self}.isGenerating)",
    cannotShowAnother: "$compute:fluid.demos.imageGenerator.cannotShowAnother({self}.state.images, {self}.state.isGenerating)",
    buttonLabel: "$compute:fluid.demos.noteImageFinder.buttonLabel({self}.isGenerating)",
    distanceLine: {
        $component: {
            $layers: "fluid.component",
            // Long-form $if so that a distance of 0 still shows the line
            $if: {
                func: distance => distance !== null,
                args: "{noteImageFinder}.distance"
            },
            view: {
                $component: {
                    $layers: "fluid.templateViewComponent",
                    template: `<div class="noteimagefinder-distance">Note is @{{noteImageFinder}.distance}px away - "@{{noteImageFinder}.closest.text}"</div>`
                }
            }
        }
    },
    errorMessage: {
        $component: {
            $layers: "fluid.component",
            $if: "{noteImageFinder}.error",
            view: {
                $component: {
                    $layers: "fluid.templateViewComponent",
                    template: `<p class="noteimagefinder-error">@{{noteImageFinder}.error}</p>`
                }
            }
        }
    },
    loadingMessage: {
        $component: {
            $layers: "fluid.component",
            $if: "{noteImageFinder}.isGenerating",
            view: {
                $component: {
                    $layers: "fluid.templateViewComponent",
                    template: `<p class="noteimagefinder-loading">Finding your image...</p>`
                }
            }
        }
    },
    result: {
        $component: {
            $layers: "fluid.component",
            $if: "{noteImageFinder}.image",
            view: {
                $component: {
                    $layers: "fluid.templateViewComponent",
                    template: `<figure class="noteimagefinder-result">
                        <img class="noteimagefinder-image" src="@{{noteImageFinder}.image.url}" alt="@{{noteImageFinder}.image.tags}"
                             @onerror="{noteImageFinder}.refresh()">
                        <figcaption class="noteimagefinder-credit">Image by <a href="@{{noteImageFinder}.image.userURL}"
                            target="_blank" rel="noopener noreferrer">@{{noteImageFinder}.image.user}</a> from <a
                            href="@{{noteImageFinder}.image.pageURL}" target="_blank" rel="noopener noreferrer">Pixabay</a></figcaption>
                    </figure>`
                }
            }
        }
    },
    generate: {
        $method: {
            // The search's promise is returned so that everything it causes is a single undoable step
            func: "fluid.demos.noteImageFinder.generate",
            args: "{self}"
        }
    },
    showAnother: {
        $method: {
            func: "fluid.demos.imageGenerator.showAnother",
            args: ["{self}.state"]
        }
    },
    refresh: {
        $method: {
            func: "fluid.demos.imageGenerator.refresh",
            args: ["{self}.state"]
        }
    }
});

fluid.demos.noteImageFinder.cannotGenerate = (distance, isGenerating) => isGenerating || distance === null;

fluid.demos.noteImageFinder.buttonLabel = isGenerating => isGenerating ? "Generating..." : "Generate image";

/**
 * Find the closest sticky note, and search Pixabay for an image of it
 * @param {Component} self - The component proxy of the widget
 * @return {Promise} A promise which resolves when the search has finished
 */
fluid.demos.noteImageFinder.generate = function (self) {
    self.state.prompt = self.closest.text;
    return fluid.demos.imageGenerator.generate(self.state);
};
</script>

<template>
    <div class="noteimagefinder">
        <h3 class="noteimagefinder-title">Closest Sticky Note</h3>
        <div class="noteimagefinder-text">@{noteText}</div>
        <div @id="distanceLine"></div>
        <div class="noteimagefinder-buttons">
            <button class="noteimagefinder-button" disabled="@{cannotGenerate}"
                    @onclick="{self}.generate()">@{buttonLabel}</button>
            <button class="noteimagefinder-button noteimagefinder-another" disabled="@{cannotShowAnother}"
                    @onclick="{self}.showAnother()">Another</button>
        </div>
        <div @id="errorMessage"></div>
        <div @id="loadingMessage"></div>
        <div @id="result"></div>
    </div>
</template>
