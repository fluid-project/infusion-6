fluid.def("fluid.demos.ai.noteImageGenerator", {
    $layers: ["fluid.demos.imageGenerator", "fluid.topLevelWidget"],
    // The closest-note finder runs hidden as a subcomponent, sharing this widget's view (and so its
    // place on the canvas), supplying the image prompt from the nearest sticky note
    finder: {
        $component: {
            $layers: "fluid.demos.closestNoteFinder",
            view: "{noteImageGenerator}.view"
        }
    },
    // The text of the closest note, left blank while no note is actually nearby, so that the Find
    // Image button is enabled only when there is something worth searching for
    closestPrompt: {
        $compute: function (text, distance) {
            return distance === null ? "" : text;
        },
        args: ["{finder}.closest.text", "{finder}.closest.distance"]
    },
    // Wire the finder's output into the generator's prompt - the input then stays writable and is
    // re-aimed at the closest note whenever the widget is moved or a note changes
    state: {
        prompt: "{noteImageGenerator}.closestPrompt"
    }
});
