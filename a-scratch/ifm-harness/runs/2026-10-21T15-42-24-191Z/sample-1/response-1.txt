// Image generator composed with the closest-note finder: this layer IS the image generator, with the finder
// hidden inside it. The finder's computation runs against this shape's bounds (it shares the view, and so the
// hosting shape), and the text of the nearest sticky note is supplied as the generator's prompt - so the user
// drags the combined widget next to a note, watches the prompt update to that note's text, and presses the
// existing Find Image button to search for it.
fluid.def("fluid.demos.ai.imageFromClosestNote", {
    $layers: ["fluid.demos.imageGenerator", "fluid.topLevelWidget"],
    finder: {
        $component: {
            $layers: "fluid.demos.closestNoteFinder",
            view: "{imageFromClosestNote}.view"
        }
    },
    state: {
        prompt: "{finder}.noteText"
    }
});
