fluid.def("fluid.demos.ai.closestNoteImage", {
    // Combined widget: the image generator is dominant and visible; the closest-note finder is a supplier
    // hidden inside it. Move the widget close to a sticky note and press "Find Image" to generate an image
    // of that note's text.
    $layers: ["fluid.demos.imageGenerator", "fluid.topLevelWidget"],
    // Hidden supplier: computes the text of the nearest sticky note from the widget's placed bounds
    finder: {
        $component: {
            $layers: "fluid.demos.closestNoteFinder",
            view: "{closestNoteImage}.view"
        }
    },
    // The generator's prompt input takes the nearest note's text (it may still be typed over)
    state: {
        prompt: "{finder}.noteText"
    }
});
