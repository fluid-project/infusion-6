// Composes the closest-note finder (supplier) with the image generator (dominant):
// wherever the widget is placed, its prompt is kept in step with the text of the
// nearest sticky note, so pressing "Find Image" searches Pixabay for it.
fluid.def("fluid.demos.ai.noteImageGenerator", {
    $layers: ["fluid.demos.imageGenerator", "fluid.topLevelWidget"],
    finder: {
        $component: {
            $layers: "fluid.demos.closestNoteFinder",
            // Share the umbrella's view: the finder renders nothing of its own,
            // but keeps computing its bounds against this shape
            view: "{noteImageGenerator}.view"
        }
    },
    state: {
        prompt: "{finder}.closest.text"
    }
});
