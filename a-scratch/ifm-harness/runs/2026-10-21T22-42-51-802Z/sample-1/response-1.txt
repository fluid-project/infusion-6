<script>
// Combines the closest-note finder with the image generator: the finder (top) watches the notes on the canvas and
// its closest note's text feeds the generator's prompt (bottom), so moving the widget next to a sticky note and
// pressing its button generates an image from that note.

fluid.def("fluid.demos.closestNoteImageGen", {
    $layers: "fluid.topLevelWidget",
    preferredSize: {
        w: 360,
        h: 680
    },
    noteFinder: {
        $component: {
            $layers: "fluid.demos.closestNoteFinder",
            // Forward the hosting context from this umbrella widget down into the nested finder
            hostBounds: "{self}.hostBounds",
            tldrawHost: "{tldrawHost}"
        }
    },
    imageGenerator: {
        $component: {
            $layers: "fluid.demos.imageGenerator",
            // The generator's prompt now tracks the closest note; it can still be edited by typing
            state: {
                prompt: "{noteFinder}.noteText"
            }
        }
    }
});
</script>

<template>
    <div class="noteimg">
        <div @id="noteFinder" class="noteimg-finder"></div>
        <div class="noteimg-gap"></div>
        <div @id="imageGenerator" class="noteimg-generator"></div>
    </div>
</template>
