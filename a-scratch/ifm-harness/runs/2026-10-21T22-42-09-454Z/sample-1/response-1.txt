<script>
// Umbrella widget combining fluid.demos.closestNoteFinder and fluid.demos.imageGenerator.
// The note finder is embedded as a subcomponent whose own view shows the closest sticky note and its distance;
// its host bounds are mirrored from this hosted widget so it keeps tracking the canvas as the whole card moves.
// The image generator is embedded below it, and its "Generate image from note" button seeds the prompt with the
// closest note's text and starts the search. The user may still type their own prompt and use the image
// generator's own button.

fluid.def("fluid.demos.ai.noteToImage", {
    $layers: "fluid.topLevelWidget",
    preferredSize: {
        w: 400,
        h: 700
    },
    finder: {
        $component: {
            $layers: "fluid.demos.closestNoteFinder",
            hostBounds: "{self}.hostBounds"
        }
    },
    imageGen: {
        $component: {
            $layers: "fluid.demos.imageGenerator"
        }
    },
    // True while the widget is away from any canvas note, so the generate-from-note button is disabled
    noNearbyNote: "$compute:fluid.demos.ai.noteToImage.hasNoNote({finder}.closest)",
    generateFromNote: {
        $method: {
            func: "fluid.demos.ai.noteToImage.generateFromNote",
            args: ["{imageGen}", "{finder}.closest"]
        }
    }
});

fluid.demos.ai.noteToImage.hasNoNote = closest => !(closest && closest.distance !== null);

fluid.demos.ai.noteToImage.generateFromNote = function (imageGen, closest) {
    if (closest && closest.distance !== null) {
        // Seed the prompt with the note's text ("sticky note" for an empty note) and search Pixabay for it
        imageGen.state.prompt = closest.text || "sticky note";
        return imageGen.generate();
    } else {
        imageGen.state.prompt = "";
        imageGen.state.error = "Move me next to a sticky note first";
        return undefined;
    }
};
</script>

<template>
    <div class="noteToImage">
        <div class="noteToImage-finderWrap" @id="finder"></div>
        <div class="noteToImage-actions">
            <button class="noteToImage-button" disabled="@{noNearbyNote}"
                    @onclick="{self}.generateFromNote()">Generate image from note</button>
            <span class="noteToImage-hint">Move me next to a sticky note, then press the button</span>
        </div>
        <div class="noteToImage-imagegenWrap" @id="imageGen"></div>
    </div>
</template>
